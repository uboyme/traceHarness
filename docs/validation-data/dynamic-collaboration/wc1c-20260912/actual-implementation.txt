def recommend(stock, target, pack, blocked=False):
    # Validate inputs per rules.md lines 1-3
    if stock < 0 or target < 0:
        raise ValueError("stock and target must be non-negative integers")
    if pack <= 0:
        raise ValueError("pack must be a positive integer")
    
    # Apply blocked flag per rule 4
    if blocked:
        return 0
    
    # Return zero when stock already reaches target per rule 5
    if stock >= target:
        return 0
    
    # Calculate units needed
    needed = target - stock
    
    # Calculate smallest whole number of packs that reaches or exceeds target per rule 6
    # This is ceiling division: (needed + pack - 1) // pack
    packs_needed = (needed + pack - 1) // pack
    
    # Return ordered unit count, not number of packs per rule 7
    return packs_needed * pack
